
CREATE TABLE Employee1(
    emp_id INT PRIMARY KEY,
    name VARCHAR(50),
    department VARCHAR(50),
    salary INT,
    age INT,
    city VARCHAR(50),
    manager_id INT,
    joining_date DATE
);
